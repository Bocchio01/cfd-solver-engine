16{

18

22 cfd->engine->method->iteractions < cfd->engine->method->maxIter;
23 cfd->engine->method->iteractions++)
24 {
25 CFD_SCGS_Reset(scgs);
26

165void CFD_SCGS_System_Compose(CFD_t *cfd, SCGS_t *scgs)
166{
167 double Ap[4];
168
169 uint16_t i = cfd->engine->method->index->i;
170 uint16_t j = cfd->engine->method->index->j;
171
172 SCGS_position positions[] = {
173 {-1, +0, u},
174 {+0, +0, u},
175 {+0, -1, v},
176 {+0, +0, v}};
177
178 for (int el = 0; el < 4; el++)
179 {
180 scgs->vanka->R->data[el] = 0.0;
181
182 // Use UDS near boundaries to avoid out-of-bounds access with QUICK scheme
183 // QUICK requires access to nodes at offset ±2, so we need 2-cell buffer
184 if (i <= cfd->engine->mesh->n_ghosts + 1 ||
185 j <= cfd->engine->mesh->n_ghosts + 1 ||
186 i >= cfd->engine->mesh->nodes->Nx + cfd->engine->mesh->n_ghosts - 2 ||
187 j >= cfd->engine->mesh->nodes->Ny + cfd->engine->mesh->n_ghosts - 2)
188 {
189 CFD_Scheme_Convection_UDS(cfd, i + positions[el].i, j + positions[el].j, positions[el].phi);
190 CFD_Scheme_Diffusion_SECOND(cfd);
191 }
192 else
193 {
194 cfd->engine->schemes->convection->callable(cfd, i + positions[el].i, j + positions[el].j, positions[el].phi);
195 cfd->engine->schemes->diffusion->callable(cfd);
196 }
197
198 for (uint16_t k = 0; k < scgs->A_coefficients->length; k++)
199 {
200 double phi = CFD_Get_State(cfd, positions[el].phi, i + positions[el].i + ((k / 5) % 5) - 2, j + positions[el].j + (k % 5) - 2);
201 scgs->A_coefficients->data[k] = cfd->engine->schemes->convection->coefficients->data[k] + cfd->engine->schemes->diffusion->coefficients->data[k];
202
203 if (k == PP)
204 {
205 Ap[el] = scgs->A_coefficients->data[k];
206 scgs->vanka->R->data[el] -= (scgs->A_coefficients->data[k]) * phi;
207 }
208 else
209 {
210 scgs->vanka->R->data[el] += (scgs->A_coefficients->data[k]) * phi;
211 }
212 }
213 }
214
215 scgs->vanka->R->data[0] += (CFD_Get_State(cfd, p, i - 1, j + 0) - CFD_Get_State(cfd, p, i + 0, j + 0)) * cfd->engine->mesh->element->size->dy;
216 scgs->vanka->R->data[1] += (CFD_Get_State(cfd, p, i + 0, j + 0) - CFD_Get_State(cfd, p, i + 1, j + 0)) * cfd->engine->mesh->element->size->dy;
217 scgs->vanka->R->data[2] += (CFD_Get_State(cfd, p, i + 0, j - 1) - CFD_Get_State(cfd, p, i + 0, j + 0)) * cfd->engine->mesh->element->size->dx;
218 scgs->vanka->R->data[3] += (CFD_Get_State(cfd, p, i + 0, j + 0) - CFD_Get_State(cfd, p, i + 0, j + 1)) * cfd->engine->mesh->element->size->dx;
219 scgs->vanka->R->data[4] = -((CFD_Get_State(cfd, u, i + 0, j + 0) - CFD_Get_State(cfd, u, i - 1, j + 0)) * cfd->engine->mesh->element->size->dy +
220 (CFD_Get_State(cfd, v, i + 0, j + 0) - CFD_Get_State(cfd, v, i + 0, j - 1)) * cfd->engine->mesh->element->size->dx);
221
222 scgs->vanka->A->data[0][0] = Ap[0] / scgs->under_relaxation->u;
223 scgs->vanka->A->data[1][1] = Ap[1] / scgs->under_relaxation->u;
224 scgs->vanka->A->data[2][2] = Ap[2] / scgs->under_relaxation->v;
225 scgs->vanka->A->data[3][3] = Ap[3] / scgs->under_relaxation->v;
226
227 scgs->vanka->A->data[4][0] = -1.0 * cfd->engine->mesh->element->size->dy;
228 scgs->vanka->A->data[4][1] = +1.0 * cfd->engine->mesh->element->size->dy;
229 scgs->vanka->A->data[4][2] = -1.0 * cfd->engine->mesh->element->size->dx;
230 scgs->vanka->A->data[4][3] = +1.0 * cfd->engine->mesh->element->size->dx;

234 scgs->vanka->A->data[2][4] = +1.0 * cfd->engine->mesh->element->size->dx;
235 scgs->vanka->A->data[3][4] = -1.0 * cfd->engine->mesh->element->size->dx;

246void CFD_SCGS_System_Solve(SCGS_t *scgs)
247{
248 double DEN;
249 double r1;
250 double r2;
251 double r3;
252 double r4;
253
254 r1 = scgs->vanka->A->data[4][0] / scgs->vanka->A->data[0][0];
255 r2 = scgs->vanka->A->data[4][1] / scgs->vanka->A->data[1][1];
256 r3 = scgs->vanka->A->data[4][2] / scgs->vanka->A->data[2][2];
257 r4 = scgs->vanka->A->data[4][3] / scgs->vanka->A->data[3][3];
258
259 DEN = r1 * scgs->vanka->A->data[0][4] +
260 r2 * scgs->vanka->A->data[1][4] +
261 r3 * scgs->vanka->A->data[2][4] +
262 r4 * scgs->vanka->A->data[3][4];
263
264 scgs->vanka->x->data[4] = (r1 * scgs->vanka->R->data[0] +
265 r2 * scgs->vanka->R->data[1] +
266 r3 * scgs->vanka->R->data[2] +
267 r4 * scgs->vanka->R->data[3] -
268 scgs->vanka->R->data[4]) /
269 DEN;
270
271 scgs->vanka->x->data[0] = (scgs->vanka->R->data[0] - scgs->vanka->A->data[0][4] * scgs->vanka->x->data[4]) / scgs->vanka->A->data[0][0];
272 scgs->vanka->x->data[1] = (scgs->vanka->R->data[1] - scgs->vanka->A->data[1][4] * scgs->vanka->x->data[4]) / scgs->vanka->A->data[1][1];
273 scgs->vanka->x->data[2] = (scgs->vanka->R->data[2] - scgs->vanka->A->data[2][4] * scgs->vanka->x->data[4]) / scgs->vanka->A->data[2][2];
274 scgs->vanka->x->data[3] = (scgs->vanka->R->data[3] - scgs->vanka->A->data[3][4] * scgs->vanka->x->data[4]) / scgs->vanka->A->data[3][3];
275}
