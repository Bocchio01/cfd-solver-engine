41 // Convergence monitor (same norm as SCGS and as Eq. 161 of the report):
42 // sum over the whole domain of the absolute value of the (signed) residuals.
43 double residual_u = 0.0;
44 double residual_v = 0.0;
45 double residual_p = 0.0;
46
47 for (uint16_t j = cfd->engine->mesh->n_ghosts;
48 j < cfd->engine->mesh->nodes->Ny + cfd->engine->mesh->n_ghosts;
49 j++)
50 {
51 for (uint16_t i = cfd->engine->mesh->n_ghosts;
52 i < cfd->engine->mesh->nodes->Nx + cfd->engine->mesh->n_ghosts;
53 i++)
54 {
55 residual_u += fabs(simple->residual->u->data[j][i]);
56 residual_v += fabs(simple->residual->v->data[j][i]);
57 residual_p += fabs(simple->residual->p->data[j][i]);
58 }
59 }
60
61 // fmax() silently discards a NaN, so a diverged field would look "converged" (residual = 0):
62 // the NaN is propagated explicitly.
63 if (isnan(residual_u) || isnan(residual_v) || isnan(residual_p))
64 {
65 cfd->engine->method->residual->data[cfd->engine->method->iteractions] = NAN;
66 }
67 else
68 {
69 cfd->engine->method->residual->data[cfd->engine->method->iteractions] = fmax(fmax(residual_u, residual_v), residual_p);
70 }
71
72 // printf("Residual: %.10f\n", cfd->engine->method->residual->data[cfd->engine->method->iteractions]);
73
74 if (isnan(cfd->engine->method->residual->data[cfd->engine->method->iteractions]) ||
75 isinf(cfd->engine->method->residual->data[cfd->engine->method->iteractions]))
76 {
77 log_fatal("Algorithm diverged at iteration %d", cfd->engine->method->iteractions);
78 CFD_SIMPLE_Free(simple);
79 exit(EXIT_FAILURE);
80 }
81
82 if (cfd->engine->method->iteractions % 100 == 0 ||
83 cfd->engine->method->residual->data[cfd->engine->method->iteractions] < cfd->engine->method->tolerance)
84 {
85 log_info("\nIteration:\t%d\nResidual:\t%.10f",
86 cfd->engine->method->iteractions,
87 cfd->engine->method->residual->data[cfd->engine->method->iteractions]);
88 }
89
90 if (cfd->engine->method->residual->data[cfd->engine->method->iteractions] < cfd->engine->method->tolerance &&
91 cfd->engine->method->iteractions > 1)
92 {
93 log_info("Algorithm converged in %d iterations", cfd->engine->method->iteractions);
94 CFD_SIMPLE_Free(simple);
95 break;
96 }
97 }
